cmake_minimum_required(VERSION 3.22)
project(usb-boop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets DBus Network Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
set(USB_BOOP_VERSION "0.0.0-dev" CACHE STRING "CalVer release version")
option(USB_BOOP_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(USB_BOOP_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
add_subdirectory(src)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
include(packaging.cmake)
